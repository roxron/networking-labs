#ISP100PE5
ip prefix-list CSC-ISP102-TO-BGP permit 10.102.0.0/16 ge 32
ip prefix-list CSC-BGP-TO-ISP102 permit 10.101.0.0/16 ge 32
!
route-map CSC-ISP102-TO-BGP permit 10
 match ip address prefix-list CSC-ISP102-TO-BGP
!
route-map CSC-BGP-TO-ISP102 permit 10
 match ip address prefix-list CSC-BGP-TO-ISP102
!
vrf definition ISP102
 route-target import 101:101
!
interface lo102
 description CsC endpoint for ISP102
 vrf forwarding ISP102
 ip address 10.100.11.5 255.255.255.255
 ip router isis ISP102
 mpls ip
!
interface g0/2
 ip router isis ISP102
 isis network point-to-point
 mpls ip
!
interface g0/3
 ip router isis ISP102
 isis network point-to-point
 mpls ip
!
router isis ISP102
 vrf ISP102
 net 49.0100.B100.0000.0005.00
 is-type level-2-only
 metric-style wide
 passive-interface lo102
 redistribute bgp 100 ip level-2 route-map CSC-BGP-TO-ISP102
 log-adjacency-changes
!
router bgp 100
 address-family ipv4 vrf ISP102
  redistribute isis ISP102 level-2 route-map CSC-ISP102-TO-BGP
 exit-address-family
!
