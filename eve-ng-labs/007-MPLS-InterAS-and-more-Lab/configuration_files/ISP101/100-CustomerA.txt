#ISP101PE115
vrf definition CustomerA
 rd 10.101.11.115:11
 !
 address-family ipv4
  route-target export 11:11
  route-target import 11:11
 exit-address-family
!
interface Loopback11
 description CustomerA BGP and OSPF endpoint
 vrf forwarding CustomerA
 ip address 10.101.11.115 255.255.255.255
 ip ospf 11 area 0
!
interface GigabitEthernet0/10.12
 description CustomerA link to CE101V133 GigabitEthernet0/0.12
 encapsulation dot1Q 12
 vrf forwarding CustomerA
 ip address 10.11.64.2 255.255.255.252
 ip ospf network point-to-point
 ip ospf 11 area 0
 no shutdown
!
router ospf 11 vrf CustomerA
 router-id 10.101.11.115
 passive-interface Loopback11
!
router bgp 101
 address-family ipv4 vrf CustomerA
  neighbor 10.11.65.133 remote-as 65511
  neighbor 10.11.65.133 description CustomerA CE101V133
  neighbor 10.11.65.133 ebgp-multihop 2
  neighbor 10.11.65.133 update-source Loopback11
  neighbor 10.11.65.133 activate
 exit-address-family
!
