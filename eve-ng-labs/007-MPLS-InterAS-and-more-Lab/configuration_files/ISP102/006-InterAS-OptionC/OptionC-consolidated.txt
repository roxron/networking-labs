#ISP102RR1
ip extcommunity-list standard RT-CUSTOMERA permit rt 11:11
!
route-map VPNV4-CUSTOMERA permit 10
 match extcommunity RT-CUSTOMERA
!
router bgp 102
 neighbor 10.101.12.1 remote-as 101
 neighbor 10.101.12.1 description InterAS Option C to ISP101RR1
 neighbor 10.101.12.1 update-source Loopback0
 neighbor 10.101.12.1 ebgp-multihop 10
 !
 address-family vpnv4
  neighbor 10.101.12.1 activate
  neighbor 10.101.12.1 send-community extended
  neighbor 10.101.12.1 next-hop-unchanged
  neighbor 10.101.12.1 route-map VPNV4-CUSTOMERA in
  neighbor 10.101.12.1 route-map VPNV4-CUSTOMERA out
 exit-address-family
!
