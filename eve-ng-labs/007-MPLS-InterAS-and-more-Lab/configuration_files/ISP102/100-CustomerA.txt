#ISP102PE25
vrf definition CustomerA
 rd 10.102.11.25:11
 !
 address-family ipv4
  route-target export 11:11
  route-target import 11:11
 exit-address-family
!
interface Loopback11
 description CustomerA BGP and OSPF endpoint
 vrf forwarding CustomerA
 ip address 10.102.11.25 255.255.255.255
 ip ospf 11 area 0
!
interface GigabitEthernet0/10.12
 description CustomerA link to CE101V131 GigabitEthernet0/0.12
 encapsulation dot1Q 12
 vrf forwarding CustomerA
 ip address 10.11.0.2 255.255.255.252
 ip ospf network point-to-point
 ip ospf 11 area 0
 no shutdown
!
router ospf 11 vrf CustomerA
 router-id 10.102.11.25
 passive-interface Loopback11
!
router bgp 102
 address-family ipv4 vrf CustomerA
  neighbor 10.11.1.131 remote-as 65511
  neighbor 10.11.1.131 description CustomerA CE101V131
  neighbor 10.11.1.131 ebgp-multihop 2
  neighbor 10.11.1.131 update-source Loopback11
  neighbor 10.11.1.131 activate
 exit-address-family
!
