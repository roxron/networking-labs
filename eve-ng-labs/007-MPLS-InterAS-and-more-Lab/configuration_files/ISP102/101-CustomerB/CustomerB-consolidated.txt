#ISP102PE25.cfg
vrf definition CustomerB
 rd 10.102.11.25:12
 !
 address-family ipv4
  route-target export 12:12
  route-target import 12:12
 exit-address-family
!
interface Loopback12
 description CustomerB BGP and OSPF endpoint
 vrf forwarding CustomerB
 ip address 10.102.11.25 255.255.255.255
 ip ospf 12 area 0
!
interface GigabitEthernet0/10
 no shutdown
!
interface GigabitEthernet0/10.13
 description CustomerB link to CE101V131 GigabitEthernet0/0.13
 encapsulation dot1Q 13
 vrf forwarding CustomerB
 ip address 10.12.0.2 255.255.255.252
 ip ospf network point-to-point
 ip ospf 12 area 0
 no shutdown
!
router ospf 12 vrf CustomerB
 router-id 10.102.11.25
 passive-interface Loopback12
!
router bgp 102
 address-family ipv4 vrf CustomerB
  neighbor 10.12.1.131 remote-as 65512
  neighbor 10.12.1.131 description CustomerB CE101V131
  neighbor 10.12.1.131 ebgp-multihop 2
  neighbor 10.12.1.131 update-source Loopback12
  neighbor 10.12.1.131 activate
  neighbor 10.12.1.131 as-override
 exit-address-family
!
